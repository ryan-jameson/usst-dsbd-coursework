package 上机作业1;

public class For {
	public static void main() {
		int n=0;
		long sum=0;
		for(int i=1;i<=100;i++) {
			if(i%9==0) {
				System.out.println(i+"是9的倍数");
				sum+=i;
				n++;
			}
		}
		System.out.println("有"+n+"个9的倍数");
		System.out.println("他们的总和是"+sum);
	}
}
