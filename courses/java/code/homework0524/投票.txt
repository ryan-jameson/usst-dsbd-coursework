package homework0524;

import java.util.Scanner;

public class Toupiao {
	public static void main(String[] args) {
		System.out.println("请输入0-9的数字，-1代表结束");
		Scanner sc=new Scanner(System.in);
		int[] arr=new int[10];
		int a=sc.nextInt();
		
		while(a!=-1) {
			for(int i=0;i<=9;i++) {
				if(i==a) arr[i]++;
			}
			a=sc.nextInt();
		}
		sc.close();
		
		for(int i=0;i<=9;i++) {
			System.out.print(i+"有"+arr[i]+"个"+"\t");
		}
	}
}
