package homework0524;

import java.util.Scanner;

public class Kuorong {
	public static void main(String[] args) {
		int[] arr= {1,2,3};
		boolean ans=false;
		Scanner sc=new Scanner(System.in);
		int k=0;
		
		do {
			int[] arr_new=new int[arr.length+1];
			for(int i=0;i<arr.length;i++) {
				arr_new[i]=arr[i];
			}
			
			if(ans!=true) {
				arr_new[arr.length]=4;
				arr=arr_new;
			}else {
				arr_new[arr.length]=k;
				arr=arr_new;
			}
			
			System.out.println("是否想要再扩容，回答true or false以及数字");
			ans=sc.nextBoolean();
			if(ans) k=sc.nextInt();
		}while(ans);
		sc.close();
		
		for(int i=0;i<arr.length;i++) {
			System.out.print(arr[i]+"\t");
		}
	}
}
