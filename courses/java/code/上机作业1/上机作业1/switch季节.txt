package 上机作业1;

import java.util.Scanner;

public class SWITCH {
	public static void main(String args[]) {
		System.out.println("请输入指定月份（1-12）");
		Scanner sc=new Scanner(System.in);
		int month=sc.nextInt();
		String result="null";
		sc.close();
		
		switch(month) {
			case 3:
			case 4:
			case 5:
				result="春季";
				break;
			case 6:
			case 7:
			case 8:
				result="夏季";
				break;
			case 9:
			case 10:
			case 11:
				result="秋季";
				break;
			case 12:
			case 1:
			case 2:
				result="冬季";
				break;
			default:
				result="输入错误，请重新输入";
		}
		System.out.println(result);
	}
}
