package homework0409;

import java.util.Scanner;

public class Switch2 {
	public static void main(String args[]) {
		System.out.println("月份为");
		Scanner sc=new Scanner(System.in);
		int m=sc.nextInt();
		sc.close();
		String result="null";
		
		switch(m) {
			case 1:
			case 2:
			case 12:
				result="冬季";
				break;
			case 3:
			case 4:
			case 5:
				result="春季";
				break;
			case 6:
			case 7:
			case 8:
				result="夏季";
				break;
			case 9:
			case 10:
			case 11:
				result="秋季";
				break;
			default:
				result="错误，请输入正确范围";
		}
		System.out.println("季节为"+result);
	}
}
