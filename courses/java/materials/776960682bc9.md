          response.sendRedirect("AdminServlet");// 登录成功，转发到管理员界面

             if (find(userList,user)) {//在数据库得到的userlist寻找user，找到代表登陆成功
                  // 将用户对象存储到session中
                  HttpSession session = request.getSession();
                  session.setAttribute("user", user);
                  // 设置cookie
                  Cookie cookie = new Cookie("userId", user.getUsername());
                  cookie.setMaxAge(60 * 60 * 24); // 1天
                  cookie.setPath("/");
                  cookie.setSecure(true);
                  response.addCookie(cookie);// 登录成功，重定向到主页
                  response.sendRedirect("main.jsp");
              } else {// 登录失败，返回登录页面并显示错误消息
                  request.setAttribute("errorMessage", "用户名或密码错误");
                  request.getRequestDispatcher("/login.jsp").forward(request, response);
              }
          }
}

private void updateBrowsingHistory(User user, Product product) {
        boolean found=false;//查找标记，用来查看商品是否存在浏览历史中，默认为false
        for (BrowsingHistory existingHistory : user.getBrowsingHistories()) {
            if (existingHistory.getProduct().getId().equals(product.getId())) {
                existingHistory.incrementViewCount();
                found=true;
                break;//如果浏览历史中存在商品，则查找标记为真，将浏览次数+1
            }
        }
        if(!found) {
        BrowsingHistory browsingHistory = new BrowsingHistory(product);
        user.addBrowsingHistory(browsingHistory);//如果不存在，则直接加入浏览历史列表
        }
    }
