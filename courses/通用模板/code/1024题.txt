第六题
import random #要导入一个random的库 不然不能生成随机数
randnum=[random.randint(1,100) for _ in range(30)] 
#randint后面会跟一个范围 用来生成1到100的随机整数 后面的代码是用来设置这个列表的大小（30个）

oushu=0 #用来给偶数下标元素求和
jishu=[] #设置一个列表来存放奇数下标元素
for i in range(30):
    if i%2==0:
        oushu+=randnum[i] #如果下标是偶数 那就把这个元素加到sum里 循环结束以后sum就是求和的值
    else:
        jishu.append(randnum[i])#把下标是奇数的元素加入到列表当中

jishu_sorted=sorted(jishu,reverse=True) #这个语句可以倒序排列表 如果想顺序排就不加后面的reverse=True

print(oushu)
print(jishu_sorted)
