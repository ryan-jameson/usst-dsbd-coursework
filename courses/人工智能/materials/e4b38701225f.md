class CNNA(nn.Module):
def __init__(self):
super(CNNA, self).__init__()
# 1. 卷积层 1: 1 输入通道 (灰度图), 32 输出通道, 3x3 卷积核
self.conv1 = nn.Conv2d(1, 32, kernel_size=3, padding=1)
# 2. 卷积层 2: 32 输入通道, 64 输出通道, 3x3 卷积核
self.conv2 = nn.Conv2d(32, 64, kernel_size=3, padding=1)
self.pool = nn.MaxPool2d(kernel_size=2, stride=2)# 3. 池化层: 2x2 最大池化
# 4. 全连接层: 输入大小计算:28x28 经过两次 2x2 池化后变为 7x7。特征图数量为 64
self.fc1 = nn.Linear(64 * 7 * 7, 128)# 因此全连接层输入为 64 * 7 * 7
self.fc2 = nn.Linear(128, 10) # 10 输出类别 (0-9)
def forward(self, x):# 正向传播路径：卷积 -> ReLU -> 池化
x = self.pool(F.relu(self.conv1(x)))
x = self.pool(F.relu(self.conv2(x)))
x = x.view(-1, 64 * 7 * 7) # 展平，将多维张量转换为一维向量
x = F.relu(self.fc1(x)) # 将展平数据输入全连接层
x = self.fc2(x)
return F.log_softmax(x, dim=1) # 用于分类任务的对数 softmax 输出

class CNNC(nn.Module):
def __init__(self):
super(CNNC, self).__init__()# 父类初始化
self.conv1 = nn.Conv2d(1, 6, kernel_size=5) # C1 使用 5*5 卷积核提取 6 种特征
self.pool = nn.MaxPool2d(2, 2)# 池化层定义 S2/S4
self.conv2 = nn.Conv2d(6, 16, kernel_size=5)# C3 再提取到 16 种细微特征
self.fc1 = nn.Linear(16 * 4 * 4, 120)# 全连接层输入: 16 * 4 * 4 = 256
self.fc2 = nn.Linear(120, 84)# 设置中间层 120 神经元
self.fc3 = nn.Linear(84, 10)
def forward(self, x):
x = self.pool(F.relu(self.conv1(x)))# C1 → ReLU → S2: 28x28 → 24x24 → 12x12
x = self.pool(F.relu(self.conv2(x)))# C3 → ReLU → S4: 12x12 → 8x8 → 4x4
x = x.view(-1, 16 * 4 * 4) # 展平: (batch, 16, 4, 4) → (batch, 256)
x = F.relu(self.fc1(x))# 全连接 1 + ReLU
x = F.relu(self.fc2(x))# 全连接 2 + ReLU
x = self.fc3(x)# 全连接 3（无激活）
return F.log_softmax(x, dim=1)

self.pool = nn.MaxPool2d(2, 2)

class EarlyStopping:
def __init__(self, patience=5, min_delta=0):
self.patience = patience#当验证集 loss 在多少个 epoch 内没有改善时停止训练
self.min_delta = min_delta#以此范围内的变化被认为是极其微小的，不计入改善
self.counter = 0 #记录未改善多少个 epoch
self.best_loss = None #验证集损失的最低值，以判定最佳模型
self.early_stop = False # 早停标志，当需要停止时为 True
self.best_model_state = None # 用于保存最佳模型参数
def __call__(self, val_loss, model):
if self.best_loss is None: # 当第一次调用时（损失首次被计算时）
self.best_loss = val_loss # 把当前验证损失设为最佳值
self.best_model_state = copy.deepcopy(model.state_dict()) # 保存当前参数
elif val_loss > self.best_loss - self.min_delta: # 当验证损失没有显著改善时
self.counter += 1 # 计数器+1，当大于等于 patience 设定时触发早停
if self.counter >= self.patience:
self.early_stop = True
else: # 当显著改善时，更新最佳损失为当前验证损失，保存参数
self.best_loss = val_loss
self.best_model_state = copy.deepcopy(model.state_dict())
self.counter = 0 # 计数器重置，模型最佳状态更新
def restore_best_model(self, model):
if self.best_model_state: # 将模型恢复到最佳状态
model.load_state_dict(self.best_model_state)

def evaluate_loss(model, data_loader): # 计算验证数据集上的平均 Loss

output = model(data) # 得到模型分类结果以计算损失
total_loss += F.nll_loss(output, target, reduction='sum').item()
avg_loss = total_loss / len(data_loader.dataset)
return avg_loss # 得到平均损失
def train_model(model, train_loader, optimizer, epoch): # 训练一个 epoch，并显示进度条
model.train() # 训练模式，可以开启 dropout 与 bn 归一化

loss_list.append(loss.item()) # 逐个记录损失
train_bar.set_postfix({'loss': f'{loss.item():.4f}'})
