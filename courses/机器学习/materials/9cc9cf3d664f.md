def compute_cost(X, y, theta):
    """
    计算线性回归的损失函数（均方误差）
    
    参数:
    X : 特征矩阵 (m x n+1) - 包含m个样本，每个样本n+1个特征（含截距项）
    y : 目标向量 (m x 1) - 真实标签值
    theta : 参数向量 (n+1 x 1) - 包括截距项权重
    
    返回:
    cost : 损失函数值 - 衡量模型预测值与真实值的偏差程度
    """
    m = len(y)
    predictions = X.dot(theta)
 # 等价于θ₀+θ₁x₁+θ₂x₂+ ... +θₙxₙ
   errors=predictions-y
#计算每个样本的预测误差（残差）：预测值-真实值
    
    cost = (1/(2*m)) * np.sum(errors**2) #计算损失函数值（均方误差）
    return cost

def robust_gradient_descent(X,y,theta,alpha,iterations, max_cost=1e10):
    """
    增强鲁棒性的梯度下降算法，防止数值溢出
    参数:
    X : 特征矩阵 (m x n+1) - 包含m个样本，每个样本n+1个特征（含截距项）
    y : 目标向量 (m x 1) - 真实标签值
    theta : 初始参数向量 (n+1 x 1) - 包括截距项权重
    alpha : 学习率（步长）
    iterations : 最大迭代次数
    max_cost : 最大允许的损失值（防止溢出）
    返回:
    theta : 优化后的参数向量（如遇溢出返回最后有效值）
    cost_history : 每次迭代的损失值记录（无效迭代用NaN填充）
    converged : 是否成功收敛（False表示发生溢出）
    """
    m = len(y)   
    cost_history = np.full(iterations, np.nan)
# 初始化损失历史记录数组，使用NaN填充
    converged = True  # 收敛状态标志
    current_cost = compute_cost(X, y, theta) 
    cost_history[0] = current_cost # 记录初始损失并存储
    # 开始迭代优化
    for i in range(1, iterations): #从第1次迭代开始（第0次已存储）
        predictions = X.dot(theta)  # 预测值计算：X·θ
        errors = predictions - y # 计算预测值和梯度、残差
        gradient = (1/m) * X.T.dot(errors)       
        theta = theta - alpha * gradient# θ参数更新
        current_cost = compute_cost(X, y, theta) # 计算新损失值
        # 检查是否发生数值溢出
        if np.isinf(current_cost) or current_cost > max_cost:
            converged = False  # 如果溢出，标记为未收敛
            theta = theta + alpha * gradient #回退到上一次有效参数
            cost_history[i:] = np.nan  # 将后续无效迭代标记为NaN
            break  # 终止迭代
        cost_history[i] = current_cost  # 存储当前损失
    return theta, cost_history, converged  # 返回优化结果

def analyze_learning_rates(X,y,alphas,iterations=100, max_cost=1e10):
    """
    分析不同学习率对梯度下降收敛的影响
    参数:
    X : 特征矩阵 (m x n+1)
    y : 目标向量 (m x 1)
    alphas : 学习率列表（用于比较不同步长效果）
    iterations : 最大迭代次数（默认100次）
    max_cost : 最大允许损失值（防止数值溢出）
    返回:
    results : 包含所有结果的字典，键为学习率，值为：
        - final_cost: 最终损失值
        - converged: 是否收敛（True/False）
        - iterations_needed: 实际所需迭代次数
    """
    results = {}# 初始化结果存储字典
    plt.figure(figsize=(14, 8))# 创建画布
    for alpha in alphas:# 为每个学习率运行梯度下降
        theta_init = np.zeros(X.shape[1])  # 每次迭代重新初始化
        _, cost_history, converged = robust_gradient_descent(
            X, y, theta_init, alpha, iterations, max_cost
        )# 执行梯度下降
        valid_indices = ~np.isnan(cost_history)# 布尔掩码获取索引
        valid_cost = cost_history[valid_indices]  # 提取有效损失值        
        if len(valid_cost) > 0:# 绘制损失曲线（仅显示有效迭代部分）
            plt.plot(np.where(valid_indices)[0],valid_cost,                     label=f'α = {alpha}', linewidth=2)# x轴为实际迭代次数
            if not converged:
                plt.scatter(len(valid_cost) - 1, valid_cost[-1],                           color='red', s=80, marker='x', zorder=10)# 标记提前终止点（红色叉号）
        results[alpha] = {# 返回结果字典
            'final_cost': valid_cost[-1] if len(valid_cost) > 0 else np.inf,
            'converged': converged,
            'iterations_needed': len(valid_cost)
        }
    plt.xlabel('Iterations', fontsize=12)  # X轴迭代次数
    plt.ylabel('Cost J', fontsize=12)      # Y轴损失函数值
    plt.title('Robust Analysis: Effect of Learning Rate on Convergence', fontsize=14)
    plt.yscale('log')# 设置y轴为对数刻度（更易观察数量级变化）
    plt.grid(True, alpha=0.3)# 添加浅色网格线
    plt.legend()# 显示图例（显示学习率）
    plt.show()# 一个学习率一条线，所有学习率都以不同颜色表现在图表
    print("\n学习率对比分析:")# 创建表格对比不同学习率的结果
    print("α\t收敛情况\t最终损失\t所需迭代次数")
    for alpha in alphas:# 格式化输出结果表格
        res = results[alpha]# 调出结果在某一行
        converged = "是" if res['converged'] else "否"
        cost = res['final_cost']
        iter_needed = res['iterations_needed']
        cost_str = f"{cost:.4f}" if np.isfinite(cost) and res['converged'] else "发散"# 格式化输出        print(f"{alpha:.3f}\t{converged}\t\t{cost_str}\t{iter_needed}")
    return results  # 返回完整结果字典

class MultiLayerPerceptron:
    """
    多层感知机(MLP)实现，用于多分类任务（如手写数字识别）
    包含一个隐藏层和一个输出层，使用ReLU和Softmax激活函数
    """
    def __init__(self, input_size, hidden_size, output_size):
        """
        初始化MLP参数
        参数:
        input_size : 输入层大小 (400)
        hidden_size : 隐藏层大小 - 可调超参数（如256）
        output_size : 输出层大小 (10) - 对应0-9数字分类
        """
        # 使用正态分布随机初始化权重矩阵（输入层→隐藏层），乘以0.01缩小数值范围（防止梯度爆炸）
        self.W1 = np.random.randn(input_size, hidden_size) * 0.01  
        # 初始化隐藏层偏置项（1×hidden_size的零向量）
        self.b1 = np.zeros((1, hidden_size))          
        # 初始化权重矩阵（隐藏层→输出层）
        self.W2 = np.random.randn(hidden_size, output_size) * 0.01 
        # 初始化输出层偏置项（1×output_size的零向量）
        self.b2 = np.zeros((1, output_size))  # shape: (1, 10)
    def relu(self, Z):
        """ReLU激活函数：max(0, z)"""
        return np.maximum(0, Z)  # 返回同形状数组，负值置零
    def softmax(self, Z):
        """Softmax激活函数：将输出转换为概率分布"""
        # 数值稳定性优化：减去最大值防止指数爆炸
        exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True))  
        # 归一化计算概率分布
        return exp_Z / np.sum(exp_Z, axis=1, keepdims=True)      
    def forward(self, X):
        """
        前向传播计算
        参数:
        X : 输入特征矩阵 (m x input_size) - m个样本，每个样本400维特征
        返回:
        A2 : 输出层激活值 (m x output_size) - 每个样本的10类概率分布
        cache : 缓存中间结果用于反向传播，包含：
            Z1, A1 - 隐藏层的加权输入和激活输出
            Z2, A2 - 输出层的加权输入和激活输出
            X - 输入数据
        """
        # 第一层计算：输入层→隐藏层
        # Z1 = X·W1 + b1 (矩阵乘法 + 广播偏置)
        # X.shape = (m, 400), W1.shape = (400, hidden_size)
        Z1 = X.dot(self.W1) + self.b1  # Z1.shape = (m, hidden_size)
        A1 = self.relu(Z1)  # 应用ReLU激活函数
        # 第二层计算：隐藏层→输出层
        # Z2 = A1·W2 + b2
        # A1.shape = (m, hidden_size), W2.shape = (hidden_size, 10)
        Z2 = A1.dot(self.W2) + self.b2  # Z2.shape = (m, 10)
        A2 = self.softmax(Z2)  # 应用Softmax激活函数        
        cache = {# 缓存中间结果用于后续反向传播
            "Z1": Z1, "A1": A1,  # 隐藏层中间值
            "Z2": Z2, "A2": A2,  # 输出层中间值
            "X": X             # 输入数据
        }
        return A2, cache  # 返回输出和缓存感知机前馈计算4.交叉熵损失和梯度计算
def cross_entropy_loss(y_pred, y_true):
    """
    计算交叉熵损失函数（用于分类任务）
    参数:
    y_pred : 预测概率分布 (m x 10) - 每个样本对应0-9的10类概率（Softmax输出）
    y_true : 真实标签向量 (m,) - 每个元素是0-9的整数类别标签
    返回:
    loss : 标量 - 平均交叉熵损失值
    """
    m = y_true.shape[0]# 获取样本数量 m
    # 创建one-hot编码的真实标签矩阵
    # 创建与y_pred同形状的零矩阵
    y_one_hot = np.zeros_like(y_pred)
    # 使用numpy的花式索引为每个样本的正确类别设置1
    # np.arange(m)生成0到m-1的索引，y_true指定列索引
    y_one_hot[np.arange(m), y_true] = 1 
    y_pred = np.clip(y_pred, 1e-15, 1 - 1e-15)# 添加数值稳定性保护，将预测概率限制在[1e-15, 1-1e-15]区间，防止log(0)导致无穷大
    log_prob = np.log(y_pred)  # 计算每个样本每个类别的log概率
    # 计算交叉熵损失：点乘计算真实类别对应的log概率（因为y_one_hot只有正确类别位置为1）
    loss = -np.sum(y_one_hot * log_prob) / m  # 取负并求平均
    return loss  # 返回标量损失值
def backpropagation(cache, y_true, W2):
    """
    反向传播计算梯度（修复版）
    参数:
    cache : 前向传播中缓存的中间结果字典，包含：
        "Z1" - 隐藏层加权输入 (m x hidden_size)
        "A1" - 隐藏层激活输出 (m x hidden_size)
        "Z2" - 输出层加权输入 (m x 10)
        "A2" - 输出层概率分布 (m x 10)
        "X" - 输入数据 (m x 400)
    y_true : 真实标签向量 (m,) - 每个元素是0-9的整数类别
    W2 : 隐藏层到输出层的权重矩阵 (hidden_size x 10)
    返回:
    grads : 参数梯度字典，包含：
        "dW1" - 输入层→隐藏层权重梯度 (400 x hidden_size)
        "db1" - 隐藏层偏置梯度 (1 x hidden_size)
        "dW2" - 隐藏层→输出层权重梯度 (hidden_size x 10)
        "db2" - 输出层偏置梯度 (1 x 10)
    """
    m = y_true.shape[0]  # 例如5000个样本时 m=5000
    # 创建one-hot编码的真实标签矩阵
    # 使用cache["A2"]的形状确保维度匹配 (m x 10)
    y_one_hot = np.zeros_like(cache["A2"])  # 初始化全零矩阵
    # 使用numpy花式索引为每个样本的正确类别设置1
    y_one_hot[np.arange(m), y_true] = 1  
    # 输出层梯度计算
    # Softmax激活的交叉熵损失导数：A2 - y_one_hot
    dZ2 = cache["A2"] - y_one_hot     
    # 计算隐藏层→输出层权重梯度：A1^T · dZ2 / m
    dW2 = cache["A1"].T.dot(dZ2) / m  # shape: (hidden_size, 10)
    # 计算输出层偏置梯度：dZ2每列的平均值
    db2 = np.sum(dZ2, axis=0, keepdims=True) / m  # shape: (1, 10)
    # 误差反向传播到隐藏层：dZ2 · W2^T
    dA1 = dZ2.dot(W2.T)  # 隐藏层梯度计算
    # ReLU激活函数的导数：Z1 > 0 时导数为1，否则为0
    dZ1 = dA1 * (cache["Z1"] > 0)  # shape: (m, hidden_size)
    # 计算输入层→隐藏层权重梯度：X^T · dZ1 / m
    dW1 = cache["X"].T.dot(dZ1) / m  # shape: (400, hidden_size)
    # 计算隐藏层偏置梯度：dZ1每列的平均值
    db1 = np.sum(dZ1, axis=0, keepdims=True) / m     
    grads = {# 存储梯度结果
        "dW1": dW1,  # 输入层→隐藏层权重梯度
        "db1": db1,  # 隐藏层偏置梯度
        "dW2": dW2,  # 隐藏层→输出层权重梯度
        "db2": db2   # 输出层偏置梯度
    }
    return grads  # 返回梯度字典用于参数更新5.多层感知机训练
def train_mlp(X_train, y_train, X_val, y_val, 
              hidden_size=128,  # 隐藏层神经元数量
              learning_rate=0.1,  # 学习率
              epochs=200,  # 训练轮数
              batch_size=64):  # 批量大小
    """
    训练多层感知机（修复版）
    参数:
    X_train, y_train - 训练数据和标签 (m_train x 400, m_train)
    X_val, y_val - 验证数据和标签 (m_val x 400, m_val)
    hidden_size - 隐藏层大小 (默认128)
    learning_rate - 学习率 (默认0.1)
    epochs - 训练轮数 (默认200)
    batch_size - 批大小 (默认64)
    返回:
    model - 训练好的MLP模型
    history - 包含训练损失、准确率等的历史记录字典
    """
    n_features = X_train.shape[1]  # 特征数量 (400)
    n_classes = len(np.unique(y_train))  # 类别数 (10)
    n_samples = X_train.shape[0]  # 训练样本总数
    model =MultiLayerPerceptron(n_features, hidden_size, n_classes)# 初始化MLP模型
    history = {# 初始化训练历史记录字典
        'train_loss': [],  # 存储每轮训练损失
        'val_loss': [],    # 存储每轮验证损失
        'train_acc': [],   # 存储每轮训练准确率
        'val_acc': []      # 存储每轮验证准确率
    }
    for epoch in range(epochs):  # 迭代每个epoch
        epoch_train_loss = 0  # 累计训练损失
        epoch_train_acc = 0   # 累计训练准确率
        # 随机打乱训练数据（每个epoch重新打乱）
        indices = np.arange(n_samples)  # 生成索引
        np.random.shuffle(indices)  # 打乱索引
        X_train_shuffled = X_train[indices]# 按打乱顺序重排特征
        y_train_shuffled = y_train[indices] #按打乱顺序重排标签
        
        
        for i in range(0, n_samples, batch_size):  # 分批训练
            # 获取当前批次数据
            X_batch = X_train_shuffled[i:i+batch_size] 
            y_batch = y_train_shuffled[i:i+batch_size]              
            # 前向传播计算预测
            y_pred, cache = model.forward(X_batch)              
            # 计算交叉熵损失，加权累计损失
            loss = cross_entropy_loss(y_pred, y_batch)
            epoch_train_loss += loss * X_batch.shape[0]             
            # 反向传播计算梯度（传递当前W2参数）
            grads = backpropagation(cache, y_batch, model.W2)
            model.W1 -= learning_rate * grads["dW1"]  
# 更新输入层→隐藏层权重
            model.b1 -= learning_rate * grads["db1"]  
# 更新隐藏层偏置
            model.W2 -= learning_rate * grads["dW2"]  
# 更新隐藏层→输出层权重
            model.b2 -= learning_rate * grads["db2"]  
# 更新输出层偏置
            # 获取最大概率的类别，计算批次准确率加权累计准确率
            y_pred_labels = np.argmax(y_pred, axis=1)
            batch_acc = accuracy_score(y_batch, y_pred_labels)
            epoch_train_acc += batch_acc * X_batch.shape[0] 
        # 计算本epoch的平均训练指标，保存到历史记录
        epoch_train_loss /= n_samples  # 损失均值
        epoch_train_acc /= n_samples  # 准确率均值
        history['train_loss'].append(epoch_train_loss) 
        history['train_acc'].append(epoch_train_acc)
        y_val_pred, _ = model.forward(X_val)  
# 前向传播预测验证集
        val_loss = cross_entropy_loss(y_val_pred, y_val)  
# 计算验证损失
        y_val_pred_labels = np.argmax(y_val_pred, axis=1)  
# 获取预测标签
        val_acc = accuracy_score(y_val, y_val_pred_labels)  
# 计算验证准确率 
        history['val_loss'].append(val_loss)# 保存验证指标
        history['val_acc'].append(val_acc)
        if (epoch + 1) % 10 == 0:# 每10轮打印进度
            print(f"Epoch [{epoch+1}/{epochs}] - Train Loss: {epoch_train_loss:.4f}, "
                  f"Val Loss: {val_loss:.4f}, Train Acc: {epoch_train_acc:.4f}, "
                  f"Val Acc: {val_acc:.4f}")
    y_val_pred, _ = model.forward(X_val)  # 最终验证集预测
    y_val_pred_labels = np.argmax(y_val_pred, axis=1)  
# 获取最终预测标签
    final_val_acc = accuracy_score(y_val, y_val_pred_labels)  
# 计算最终准确率
    print(f"\nTraining completed. Final validation accuracy: {final_val_acc:.4f}") 
    return model, history  # 返回训练好的模型和历史记录6.分类正确率计算

def sigmoid(z):
    """
    计算Sigmoid函数值，支持标量、向量和矩阵输入
    参数:
    z -- 输入值，可以是标量、向量或矩阵
    数值稳定的sigmoid函数实现
    返回:
    与输入相同维度的Sigmoid计算结果
    """
    pos_mask = (z >= 0)# 对于正数输入，使用负数形式计算避免溢出
    neg_mask = (z < 0)# 对于负数则不变
    result = np.zeros_like(z)# 初始化结果数组
    # 对于正数输入使用稳定的计算方式
    result[pos_mask] = 1 / (1 + np.exp(-z[pos_mask]))
    exp_z = np.exp(z[neg_mask])# 对于负数输入，使用另一种形式计算
    result[neg_mask] = exp_z / (1 + exp_z)
    result = np.clip(result, 1e-15, 1-1e-15)# 防止除以零
    return result
2.交叉熵损失函数计算
def cross_entropy_loss(y, y_pred):
    """
    计算交叉熵损失
    参数:
        y: 真实标签 (n_samples,)
        y_pred: 预测概率 (n_samples,)
    返回:
        平均交叉熵损失
    """
    epsilon = 1e-15  # 防止log(0)出现
    y_pred=np.clip(y_pred, epsilon, 1 - epsilon) # 限制预测值范围
    return -np.mean(y * np.log(y_pred) + (1 - y) * np.log(1 - y_pred))
3.梯度计算
def compute_gradients(X, y, y_pred):
    """
    计算损失函数关于权重和偏置的梯度
    参数:
        X: 特征矩阵 (n_samples, n_features)
        y: 真实标签 (n_samples,)
        y_pred: 预测概率 (n_samples,)
    返回:
        dw: 权重梯度 (n_features,)
        db: 偏置梯度 (标量)
    """
    m = len(y)# 样本数量
    error = y_pred - y  # 预测值与真实值的差
    dw = (1/m) * np.dot(X.T, error)  # 权重梯度
    db = (1/m) * np.sum(error)  # 偏置梯度
    return dw, db

def gradient_descent(X, y, learning_rate, n_iters):
    """
    使用梯度下降优化逻辑回归模型
    参数:
        X: 特征矩阵 (n_samples, n_features) - 样本数×特征数的二维数组
        y: 标签向量 (n_samples,) - 一维数组，元素为0或1（二分类）
        learning_rate: 学习率（控制参数更新步长）
        n_iters: 迭代次数（训练轮数）
    返回:
        w: 优化后的权重 (n_features,) - 每个特征对应的系数
        b: 优化后的偏置 (标量) - 决策边界的截距项
        losses: 每次迭代的损失值列表 - 用于监控收敛过程
    """
    m, n = X.shape  # m=样本数量, n=特征维度
    w = np.zeros(n)  # 权重初始化为零向量
    b = 0.0          # 偏置初始化为0
    losses = []  # 用于存储每次迭代的损失值
    for i in range(n_iters):  # 进行n_iters次参数更新
        linear = np.dot(X, w) + b  # 计算线性组合输出（logit）
        y_pred = sigmoid(linear)  # 应用sigmoid函数转换为概率
        # 使用交叉熵损失函数评估预测质量
        loss = cross_entropy_loss(y, y_pred)  # 标量值
        losses.append(loss)  # 记录当前损失
        # 计算梯度（根据交叉熵损失对参数的导数）
        dw, db = compute_gradients(X, y, y_pred)  # 反向传播
        w -= learning_rate * dw  # 权重更新
        b -= learning_rate * db  # 偏置更新
        if i % 100 == 0:  # 每100次迭代输出一次损失
            print(f"Iteration {i}/{n_iters}, Loss: {loss:.4f}")
    print(f"Final loss: {loss:.4f}")  # 输出最终损失
    return w, b, losses  # 返回优化后的参数和损失历史

    将二维特征映射到多项式特征空间
    参数:
        X1: 第一维特征向量
        X2: 第二维特征向量
        degree: 多项式次数
    返回:
        多项式特征矩阵
    """
    m = len(X1)
    features = np.ones((m, 1))# 初始化特征矩阵，第一列为1（偏置项）
    for i in range(1, degree+1):# 生成多项式特征
        for j in range(0, i+1):
            new_feature = (X1**(i-j)) * (X2**j)# 特征项
            features = np.c_[features, new_feature]# 添加
    return features
8.带正则化的损失与梯度
def regularized_cross_entropy_loss(X, y, w, lambda_=1.0):
    """
    带正则化的交叉熵损失函数
    参数:
        X: 特征矩阵
        y: 标签向量
        w: 权重向量
        lambda_: 正则化参数
    返回:
        正则化损失值
    """
    m = len(y)
    linear = np.dot(X, w)
    y_pred = sigmoid(linear)# 得到预测值
    epsilon = 1e-15 # 将预测概率限制在[epsilon, 1-epsilon]区间，防止log(0)导致无穷大
    y_pred = np.clip(y_pred, epsilon, 1 - epsilon)# 计算基础损失
    loss = -np.mean(y * np.log(y_pred) + (1 - y) * np.log(1 - y_pred))
    # 添加L2正则化项（注意：通常不正则化偏置项）
    reg_loss = (lambda_ / (2 * m)) * np.sum(w[1:]**2)
    return loss + reg_loss
def regularized_gradients(X, y, w, lambda_=1.0):
    """
    计算带正则化的梯度
    参数:
        X: 特征矩阵
        y: 标签向量
        w: 权重向量
        lambda_: 正则化参数
    返回:
        梯度向量
    """
    m = len(y)
    linear = np.dot(X, w)
    y_pred = sigmoid(linear)
    error = y_pred - y # 计算基础残差
    grad = (1 / m) * np.dot(X.T, error)# 计算基础梯度
    reg_grad = (lambda_ / m) * w # 添加正则化梯度（不应用于偏置项）
    reg_grad[0] = 0  # 不对偏置项正则化
    return grad + reg_grad
9.训练集分类精度的计算
def calculate_accuracy(y_true, y_pred):
    """
    计算分类准确率
    参数:
        y_true: 真实标签
        y_pred: 预测标签
    返回:
        准确率（百分比）
    """
    correct=np.sum(y_true == y_pred)# 如果真实值和预测值相同则正确
    total = len(y_true)# 真实值的样本数量，作为分母
    return (correct / total) * 100

2.手动实现PCA
def manual_pca(X, n_components=2):
    """
    手动实现主成分分析(PCA)
    参数:
    X -- 输入数据矩阵 (m, n)，其中m是样本数，n是特征数
    n_components -- 要保留的主成分数量
    返回:
    X_pca -- 降维后的数据矩阵 (m, n_components)
    explained_variance_ratio -- 各主成分解释的方差比例
    """
    # 使用np.cov计算特征之间的协方差矩阵
    # 默认情况下，np.cov会自动对数据进行中心化（减去均值）
    cov_matrix=np.cov(X.T)# 结果为(n, n)矩阵，描述特征间的线性关系
    # eigenvalues: (n,)，eigenvectors: (n, n)，每个特征向量对应一个特征值
    eigenvalues, eigenvectors = np.linalg.eigh(cov_matrix)# 使用np.linalg.eigh（适用于对称矩阵）获取协方差矩阵的特征值和向量
    # np.argsort返回升序索引，[::-1]反转为降序，sorted_index为从大到小的特征值索引
    sorted_index = np.argsort(eigenvalues)[::-1]# 按特征值降序排列
    # 根据排序索引重新排列特征向量，列向量顺序对应特征值大小
    sorted_eigenvectors = eigenvectors[:, sorted_index]
    # 选择前n_components个特征向量（主成分），这些向量构成投影矩阵，用于将数据映射到低维空间
    top_eigenvectors = sorted_eigenvectors[:, :n_components]
    # 计算主成分并返回，将原始数据X(m,n)与投影矩阵(n_components,n)相乘，得到降维后的X_pca(m,n_components)
    X_pca = X @ top_eigenvectors
    total_variance = eigenvalues.sum()# 总方差为所有特征值之和
    # 每个主成分的方差比例 =对应特征值/总方差，取前n_components个
    explained_variance_ratio = eigenvalues[sorted_index][:n_components] / total_variance
    return X_pca, explained_variance_ratio
3.K均值聚类
def kmeans_clustering(X, n_clusters=10, max_iter=300):
    """
    K-Means聚类算法实现
    参数:
    X : 输入数据 (m, n) - m个样本，n个特征
    n_clusters : 聚类数量（默认10）
    max_iter : 最大迭代次数（默认300）
    返回:
    labels : 聚类标签 (m,) - 每个样本的簇标签
    centroids : 聚类中心点 (n_clusters, n) - 每个簇的中心位置
    """
    np.random.seed(42)  # 随机初始化聚类中心点
    centroids = X[np.random.choice(X.shape[0], n_clusters, replace=False), :]# 从数据中随机选择n_clusters个样本作为初始中心
    for _ in range(max_iter): # 计算所有样本到所有中心点的欧氏距离
        distances = np.sqrt(((X - centroids[:, np.newaxis])**2).sum(axis=2))#使用广播机制计算距离
        # 找到每个样本最近的簇索引（最小距离对应的簇）
        labels = np.argmin(distances, axis=0)  # shape=(m,)
        new_centroids = np.array([X[labels == k].mean(axis=0) for k in range(n_clusters)]) # 对每个簇计算其所有样本的均值作为新的中心点
        if np.all(centroids == new_centroids):#收敛检查
            break# 如果中心点不再变化则提前终止迭代
        centroids = new_centroids # 更新中心点
        
    return labels, centroids  # 返回聚类标签和最终中心点

def stratified_sampling(X, y, n_samples=1000):
    """
    分层抽样：从每个类别抽取相同数量的样本（保持类别分布）
    参数:
    X : 输入数据 (m x n) - m个样本，n个特征
    y : 标签向量 (m,) - 每个元素为类别标签（如0-9）
    n_samples : 总抽样数量（默认1000）
    返回:
    X_sample : 抽样后的数据 (n_samples x n)
    y_sample : 抽样后的标签 (n_samples,)
    """
    sss = StratifiedShuffleSplit(n_splits=1, test_size=n_samples, random_state=42)# n_splits=1 表示只生成一次划分
    # 执行分层抽样（实际只运行一次循环）
    for _, test_index in sss.split(X, y):  
# split()返回训练索引和测试索引
        X_sample = X[test_index]  # 使用测试索引获取抽样数据
        y_sample = y[test_index]
    return X_sample, y_sample  # 返回分层抽样结果
