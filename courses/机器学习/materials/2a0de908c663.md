3.梯度下降函数
def
robust_gradient_descent(X,y,theta,alpha,iterations,
max_cost=1e10):
"""
增强鲁棒性的梯度下降算法，防止数值溢出
参数:
X : 特征矩阵 (m x n+1) - 包含 m 个样本，每个样本 n+1 个特征（含
截距项）
y : 目标向量 (m x 1) - 真实标签值
theta : 初始参数向量 (n+1 x 1) - 包括截距项权重
alpha : 学习率（步长）
iterations : 最大迭代次数
max_cost : 最大允许的损失值（防止溢出）
返回:
theta : 优化后的参数向量（如遇溢出返回最后有效值）
cost_history : 每次迭代的损失值记录（无效迭代用 NaN 填充）
converged : 是否成功收敛（False 表示发生溢出）
"""
m = len(y)
cost_history = np.full(iterations, np.nan)
# 初始化损失历史记录数组，使用 NaN 填充
5

## 分节 6

converged = True # 收敛状态标志
current_cost = compute_cost(X, y, theta)
cost_history[0] = current_cost # 记录初始损失并存储
# 开始迭代优化
for i in range(1, iterations): #从第 1 次迭代开始（第 0 次已存储）
predictions = X.dot(theta) # 预测值计算：X·θ
errors = predictions - y # 计算预测值和梯度、残差
gradient = (1/m) * X.T.dot(errors)
theta = theta - alpha * gradient# θ参数更新
current_cost = compute_cost(X, y, theta) # 计算新损失值
# 检查是否发生数值溢出
if np.isinf(current_cost) or current_cost > max_cost:
converged = False # 如果溢出，标记为未收敛
theta = theta + alpha * gradient #回退到上一次有效参数
cost_history[i:] = np.nan # 将后续无效迭代标记为 NaN
break # 终止迭代
cost_history[i] = current_cost # 存储当前损失
return theta, cost_history, converged # 返回优化结果

def analyze_iterations(X, y, alpha, iterations_list, max_cost=1e10):
"""
分析不同迭代次数对最终损失的影响
参数:
X : 特征矩阵 (m x n+1)
y : 目标向量 (m x 1)
alpha : 学习率（步长）
iterations_list : 不同迭代次数的列表（用于比较）
max_cost : 最大允许损失值（防止数值溢出）
返回:
results : 包含所有结果的字典，包含：
- iterations: 迭代次数列表
- final_costs: 对应的最终损失值
- converged: 收敛状态（True/False）
"""
final_costs = []# 初始化存储最终损失值的列表
theta_init = np.zeros(X.shape[1])
# 初始化参数θ为 0 向量（维度等于特征数量）
converged_flags = []# 初始化收敛状态列表
# 创建画布，设置图表尺寸为 12 英寸宽×8 英寸高
plt.figure(figsize=(12, 8))
# 为每个迭代次数运行梯度下降
for max_iters in iterations_list:
_, cost_history, converged = robust_gradient_descent(
X, y, theta_init, alpha, max_iters, max_cost
)
# 获取有效损失值（排除 NaN 值）
valid_costs = cost_history[~np.isnan(cost_history)]
if len(valid_costs) > 0:
# 记录最后一个有效损失值
final_cost = valid_costs[-1]
# 记录收敛状态
converged_flags.append(converged)
else:
# 如果没有有效值（完全溢出），损失为空值，收敛状态 False
final_cost = np.nan
converged_flags.append(False)
final_costs.append(final_cost)# 将结果添加到列表中
plt.plot(iterations_list, final_costs, 'bo-', markersize=8,
label=f'α={alpha}')# 绘制迭代次数与最终损失的关系图

# 迭代次数列表

# 对应的最终损失值

二、实验原理（使用的机器学习数学原理）
1.分层抽样
用于保持数据集中各类别（如 0-9 数字）的比例在训练集和测试集中一致。
其数学基础是统计学中的抽样理论，目标是避免因随机划分导致的类别不平衡
（如测试集某类样本过少），确保模型评估的可靠性。若原数据集类别 Ck 的样
本占比为 pk，则训练集和测试集中类别 Ck 的样本数分别为 Ntrain⋅ pk 和 Ntest⋅ pk（N
train+Ntest=Ntotal）。
2.MLP 前馈神经网络
由输入层、隐藏层（可能多层）和输出层组成。前馈计算的核心是逐层的线
性变换与非线性激活，依赖线性代数（矩阵运算）和非线性函数。在隐藏层单层
前馈计算中，输入向量 x∈Rd（如 28×28 手写数字展平为 784 维向量），权重

层输出（经激活函数σ，这里使用 softmax）：a=σ(z)；多层前馈：输出层继

0-9 数字的概率）。

class MultiLayerPerceptron:
"""
多层感知机(MLP)实现，用于多分类任务（如手写数字识别）
包含一个隐藏层和一个输出层，使用 ReLU 和 Softmax 激活函数
"""
def __init__(self, input_size, hidden_size, output_size):
"""
初始化 MLP 参数
参数:
input_size : 输入层大小 (400)
hidden_size : 隐藏层大小 - 可调超参数（如 256）
output_size : 输出层大小 (10) - 对应 0-9 数字分类
"""
# 使用正态分布随机初始化权重矩阵（输入层→隐藏层），乘以 0.01
缩小数值范围（防止梯度爆炸）
self.W1 = np.random.randn(input_size, hidden_size) * 0.01
# 初始化隐藏层偏置项（1×hidden_size 的零向量）
self.b1 = np.zeros((1, hidden_size))
# 初始化权重矩阵（隐藏层→输出层）
self.W2 = np.random.randn(hidden_size, output_size) * 0.01
# 初始化输出层偏置项（1×output_size 的零向量）
self.b2 = np.zeros((1, output_size)) # shape: (1, 10)
def relu(self, Z):
"""ReLU 激活函数：max(0, z)"""
return np.maximum(0, Z) # 返回同形状数组，负值置零
def softmax(self, Z):
19

## 分节 20

"""Softmax 激活函数：将输出转换为概率分布"""
# 数值稳定性优化：减去最大值防止指数爆炸
exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True))
# 归一化计算概率分布
return exp_Z / np.sum(exp_Z, axis=1, keepdims=True)
def forward(self, X):
"""
前向传播计算
参数:
X : 输入特征矩阵 (m x input_size) - m 个样本，每个样本 400
维特征
返回:
A2 : 输出层激活值 (m x output_size) - 每个样本的 10 类概率
分布
cache : 缓存中间结果用于反向传播，包含：
Z1, A1 - 隐藏层的加权输入和激活输出
Z2, A2 - 输出层的加权输入和激活输出
X - 输入数据
"""
# 第一层计算：输入层→隐藏层
# Z1 = X·W1 + b1 (矩阵乘法 + 广播偏置)
# X.shape = (m, 400), W1.shape = (400, hidden_size)
Z1 = X.dot(self.W1) + self.b1 # Z1.shape = (m, hidden_size)
A1 = self.relu(Z1) # 应用 ReLU 激活函数
# 第二层计算：隐藏层→输出层
# Z2 = A1·W2 + b2
# A1.shape = (m, hidden_size), W2.shape = (hidden_size, 10)
Z2 = A1.dot(self.W2) + self.b2 # Z2.shape = (m, 10)
A2 = self.softmax(Z2) # 应用 Softmax 激活函数
cache = {# 缓存中间结果用于后续反向传播
"Z1": Z1, "A1": A1, # 隐藏层中间值
"Z2": Z2, "A2": A2, # 输出层中间值
"X": X
# 输入数据
}
return A2, cache # 返回输出和缓存感知机前馈计算

"Z1" - 隐藏层加权输入 (m x hidden_size)
"A1" - 隐藏层激活输出 (m x hidden_size)
"Z2" - 输出层加权输入 (m x 10)

"X" - 输入数据 (m x 400)

grads : 参数梯度字典，包含：
"dW1" - 输入层→隐藏层权重梯度 (400 x hidden_size)
"db1" - 隐藏层偏置梯度 (1 x hidden_size)
"dW2" - 隐藏层→输出层权重梯度 (hidden_size x 10)
"db2" - 输出层偏置梯度 (1 x 10)

# 使用 numpy 花式索引为每个样本的正确类别设置 1
y_one_hot[np.arange(m), y_true] = 1
# 输出层梯度计算
# Softmax 激活的交叉熵损失导数：A2 - y_one_hot
dZ2 = cache["A2"] - y_one_hot

def cross_entropy_loss(y, y_pred):
"""

def compute_gradients(X, y, y_pred):
"""
计算损失函数关于权重和偏置的梯度
参数:
X: 特征矩阵 (n_samples, n_features)
y: 真实标签 (n_samples,)
y_pred: 预测概率 (n_samples,)
返回:

db: 偏置梯度 (标量)
"""

db = (1/m) * np.sum(error) # 偏置梯度
return dw, db

def kmeans_clustering(X, n_clusters=10, max_iter=300):
"""
K-Means 聚类算法实现
参数:
X : 输入数据 (m, n) - m 个样本，n 个特征
n_clusters : 聚类数量（默认 10）
max_iter : 最大迭代次数（默认 300）
返回:
labels : 聚类标签 (m,) - 每个样本的簇标签
centroids : 聚类中心点 (n_clusters, n) - 每个簇的中心位置
"""
np.random.seed(42) # 随机初始化聚类中心点
centroids
=
X[np.random.choice(X.shape[0],
n_clusters,
replace=False), :]# 从数据中随机选择 n_clusters 个样本作为初始中心
for _ in range(max_iter): # 计算所有样本到所有中心点的欧氏距离
distances
=
np.sqrt(((X
centroids[:,
np.newaxis])**2).sum(axis=2))#使用广播机制计算距离
# 找到每个样本最近的簇索引（最小距离对应的簇）
labels = np.argmin(distances, axis=0) # shape=(m,)
new_centroids = np.array([X[labels == k].mean(axis=0) for k
in range(n_clusters)]) # 对每个簇计算其所有样本的均值作为新的中心点
if np.all(centroids == new_centroids):#收敛检查
break# 如果中心点不再变化则提前终止迭代
centroids = new_centroids # 更新中心点

return labels, centroids # 返回聚类标签和最终中心点
4.分层抽样
def stratified_sampling(X, y, n_samples=1000):
"""
分层抽样：从每个类别抽取相同数量的样本（保持类别分布）
参数:
X : 输入数据 (m x n) - m 个样本，n 个特征
y : 标签向量 (m,) - 每个元素为类别标签（如 0-9）
n_samples : 总抽样数量（默认 1000）
返回:
X_sample : 抽样后的数据 (n_samples x n)
y_sample : 抽样后的标签 (n_samples,)
"""
sss = StratifiedShuffleSplit(n_splits=1, test_size=n_samples,
random_state=42)# n_splits=1 表示只生成一次划分
# 执行分层抽样（实际只运行一次循环）
for _, test_index in sss.split(X, y):
# split()返回训练索引和测试索引
X_sample = X[test_index] # 使用测试索引获取抽样数据
y_sample = y[test_index]
return X_sample, y_sample # 返回分层抽样结果
