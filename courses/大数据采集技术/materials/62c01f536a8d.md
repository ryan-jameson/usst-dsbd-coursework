def fetch(url):
    last_error = None
    for attempt in range(3):
        try:
            response = requests.get(url, headers=HEADERS, timeout=15)
            response.raise_for_status()
            if response.text.strip():
                return response.text
            last_error = RuntimeError(f"empty response from {url}")
        except requests.RequestException as exc:
            last_error = exc
        time.sleep(1 + attempt)

    raise RuntimeError(f"failed to fetch {url}: {last_error}")
该部分代码设置了请求头和重试机制，避免由于网络波动或接口短暂空响应导致程序直接失败。
②解析 RSS 新闻列表：
def parse_rss_items(xml_text):
    root = ET.fromstring(xml_text)
    items = []

    for item in root.findall(".//channel/item"):
        title = unescape(get_text(item.find("title")))
        pub_time = get_text(item.find("pubDate"))
        link = get_text(item.find("link"))
        summary = BeautifulSoup(unescape(get_text(item.find("description"))), "html.parser").get_text(
            " ", strip=True
        )

        if title and link:
            items.append({"title": title, "time": pub_time, "url": link, "summary": summary})

    return items
这段代码从 RSS XML 中提取新闻标题、发布时间、新闻链接和摘要，为后续访问详情页做准备。
③提取新闻正文：
def parse_article_content(url, fallback):
    try:
        html = fetch(url)
    except requests.RequestException:
        return fallback

    soup = BeautifulSoup(html, "html.parser")

    for tag in soup(["script", "style", "nav", "footer", "aside"]):
        tag.decompose()

    selectors = ["article .article-body p", "article p", ".article-body p", "main p"]

    paragraphs = []
    for selector in selectors:
        paragraphs = [
            p.get_text(" ", strip=True)
            for p in soup.select(selector)
            if len(p.get_text(" ", strip=True)) >= 30
        ]
        if paragraphs:
            break

    content = "\n".join(paragraphs).strip()
    return content or fallback
该代码通过 CSS 选择器定位正文段落，并过滤过短文本，尽量减少导航栏、广告和无关内容对正文提取的影响。
④读取 Excel 并生成可视化：
def read_inline_xlsx(path):
    namespace = {"x": "http://schemas.openxmlformats.org/spreadsheetml/2006/main"}
    with zipfile.ZipFile(path) as archive:
        sheet_xml = archive.read("xl/worksheets/sheet1.xml")

    root = ET.fromstring(sheet_xml)
    rows = []
    for row in root.findall(".//x:sheetData/x:row", namespace):
        values = []
        for cell in row.findall("x:c", namespace):
            text_node = cell.find(".//x:t", namespace)
            values.append(text_node.text if text_node is not None and text_node.text else "")
        rows.append(values)

    headers = rows[0]
    return [dict(zip(headers, row)) for row in rows[1:]]
可视化脚本 `visualize_sports_news.py` 使用 `zipfile` 和 `ElementTree` 直接读取 `.xlsx` 内部 XML，避免依赖额外的 Excel 读取库。

①文本标准化：对 `City`、`Country` 去除首尾空格；将 `City` 统一为标题格式，例如 `SYDNEY` 处理为 `Sydney`。
②日期标准化：使用 `pandas.to_datetime` 将日期解析为日期类型，最终输出为 `YYYY-MM-DD`。
③类型转换：将污染物、气象字段统一转换为数值类型；无法转换的值设为缺失值。
④合法性检查：污染物浓度、风速不能为负；湿度应在 0 到 100 之间；温度使用 -90 到 60 摄氏度作为合理范围。非法值统一转为缺失值后再填补。
⑤重复处理：先删除标准化后的完全重复记录；再按 `City`、`Country`、`Date` 聚合，数值字段取中位数，得到城市每日级数据。
⑥缺失值填补：对数值字段采用分层中位数填补，优先使用“同城市同月份”中位数，其次使用“同城市”中位数，最后使用全局中位数。中位数对极端值更稳健，适合空气质量这类偏态数值。
⑦字段名规范化：将 `PM2.5` 改为 `PM2_5`，将 `Wind Speed` 改为 `Wind_Speed`，便于后续代码调用。

def normalize_text(df: pd.DataFrame) -> pd.DataFrame:
    cleaned = df.copy()
    cleaned["City"] = cleaned["City"].astype(str).str.strip().str.title()
    cleaned["Country"] = cleaned["Country"].astype(str).str.strip()
    return cleaned


def coerce_types(df: pd.DataFrame) -> pd.DataFrame:
    cleaned = df.copy()
    cleaned["Date"] = pd.to_datetime(cleaned["Date"], errors="coerce")
    for col in NUMERIC_COLUMNS:
        cleaned[col] = pd.to_numeric(cleaned[col], errors="coerce")
    return cleaned
③ 不合法值检测与处理：
def mark_invalid_values(df: pd.DataFrame) -> tuple[pd.DataFrame, dict[str, int]]:
    cleaned = df.copy()
    invalid_masks = {
        "Date": cleaned["Date"].isna(),
        "PM2.5": cleaned["PM2.5"] < 0,
        "PM10": cleaned["PM10"] < 0,
        "NO2": cleaned["NO2"] < 0,
        "SO2": cleaned["SO2"] < 0,
        "CO": cleaned["CO"] < 0,
        "O3": cleaned["O3"] < 0,
        "Temperature": (cleaned["Temperature"] < -90) | (cleaned["Temperature"] > 60),
        "Humidity": (cleaned["Humidity"] < 0) | (cleaned["Humidity"] > 100),
        "Wind Speed": cleaned["Wind Speed"] < 0,
    }

    invalid_counts = {col: int(mask.sum()) for col, mask in invalid_masks.items()}
    for col, mask in invalid_masks.items():
        cleaned.loc[mask, col] = pd.NA
    return cleaned, invalid_counts
④ 重复记录聚合：
def aggregate_duplicate_keys(df: pd.DataFrame) -> pd.DataFrame:
    return (
        df.groupby(KEY_COLUMNS, as_index=False, dropna=False)[NUMERIC_COLUMNS]
        .median(numeric_only=True)
        .sort_values(KEY_COLUMNS)
        .reset_index(drop=True)
)
⑤ 分层中位数填补缺失值：
def fill_missing_values(df: pd.DataFrame) -> tuple[pd.DataFrame, dict[str, int]]:
    cleaned = df.copy()
    missing_before = cleaned[NUMERIC_COLUMNS].isna().sum().astype(int).to_dict()
    cleaned["_Month"] = cleaned["Date"].dt.month

    for col in NUMERIC_COLUMNS:
        city_month_median = cleaned.groupby(["City", "_Month"])[col].transform("median")
        city_median = cleaned.groupby("City")[col].transform("median")
        global_median = cleaned[col].median()

        cleaned[col] = cleaned[col].fillna(city_month_median)
        cleaned[col] = cleaned[col].fillna(city_median)
        cleaned[col] = cleaned[col].fillna(global_median)

    cleaned = cleaned.drop(columns=["_Month"])
    return cleaned, missing_before

def main() -> None:

cleaned, missing_before_impute = fill_missing_values(aggregated)

    cleaned_for_csv = cleaned.rename(columns=RENAME_COLUMNS)
