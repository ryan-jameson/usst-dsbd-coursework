选做题：编程实现模式匹配的BF算法和KMP
#include <bits/stdc++.h>

#define MAXSIZE 100
typedef struct{
char data[MAXSIZE];
int length;
}SqString;

void InitSqString(SqString s){
char x;
while(s.length<=MAXSIZE){
scanf("%c",&x);
if(x!='\n'){
s.data[s.length-1]=x;
s.length++;
continue;
}
else break;
}
}

int BF(SqString s,SqString t){
int lens=s.length;
int lent=t.length;
int i=0;
int j=0;
for (int i = 0; i <= lens; i++) {
    for (int j = 0; j < lent; j++) {
            if (s.data[i + j] != t.data[j]) {
                break;
            }
        }
}
if(j==lent) return i-j;
return -1;
}//BF算法

void Getnext(SqString t,int next[]) {
int lent=t.length;
int k=0;
for(int i=0;i<=lent;i++){
if(i==0) next[i]=-1;
else{
for(int j=i-1;j>=k;j--){
if(t.data[k]==t.data[j]){
k++;
j--;
continue;
}
}
next[i]=k;
}
}
}
int KMP(SqString s,SqString t){
int next[MAXSIZE];
Getnext(s,next);
int lens=s.length;
int lent=t.length;
int i=0;
int j=0;
while(i<lens || j<lent){
if(s.data[i]==t.data[j]){
i++;
j++;
}else{
j=next[j];
if(j==-1) i++;
}
}
if(j>=lent) return i-j;
return -1;
} 

main(){
SqString s,t;
printf("输入模板串\n");
InitSqString(t);
printf("输入主串\n");
InitSqString(s);

if(BF(s,t)!=-1) printf("匹配成功！\n");
else printf("匹配失败！\n");

if(KMP(s,t)!=-1)  printf("匹配成功！\n");
else printf("匹配失败！\n");
}
