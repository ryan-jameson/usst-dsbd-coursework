1分析以下算法的时间复杂度。
void add(int n,a[N][N],b[N][N],int c[N][N])
{  int i,j;
   for (i=0;i<n;i++)//① 频度为n+1，循环体执行n次
      for (j=0;j<n;j++)//②频度为n（n+1），循环体执行n2次
      {  c[i][j]=0;//③频度为n2
         for (k=0;k<n;k++)//④频度为n2（n+1），循环体执行n3次
           c[i][j]=c[i][j]+a[i][k]*b[k][j];//⑤频度为n3
      }
}
T(n)=n+1+n（n+1）+n2+n2（n+1）+n3=2n3+3n2+2n+1=O(n3)

2、给出以下算法的时间复杂度。
void func(int n)
{  int i=1,k=100;//频度为1
   while (i<=n)//n为偶数时，频度为n/2+1，循环体执行n/2次；n为奇数时，频度为（n+1）/2+1，循环体执行（n+1）/2次
   {   k++;//n为偶数时，频度为n/2；n为奇数时，频度为（n+1）/2
       i+=2;//n为偶数时，频度为n/2；n为奇数时，频度为（n+1）/2
   }
}
n为偶数时，T(n1)=3n/2+2；n为奇数时，T(n2)=3(n+1)/2+2
T(n)=O(n)

3、简述数据类型与抽象数据类型的区别
数据类型是一个值的集合和定义在此集合上的一组操作的总称，例如int，float
抽象数据类型=逻辑结构＋抽象运算，抽象数据类型暂不考虑计算机的具体存储结构和运算的具体实现，而数据类型需要考虑，可以说抽象数据类型就是数据类型的更大范围，比如结构体student中包含学生的学号（int），姓名（char），成绩（float），而他本身作为抽象数据类型不能进行任何数学或逻辑运算，主要用于在计算机世界中解决现实问题
