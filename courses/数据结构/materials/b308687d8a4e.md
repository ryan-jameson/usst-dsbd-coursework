typedef struct{
int top;//栈顶元素 
int data[MAXSIZE];
}SqStack ,*linkStack;//栈定义

linkStack InitStack(SqStack *s){
s=(linkStack)malloc(sizeof(SqStack));
s->top=-1;//初始化栈顶 
return s;
} //栈初始化

int Push(SqStack *s,int e){
if(s->top==MAXSIZE-1) return 0;//栈满条件 
else{
s->top++;//栈顶上移 
s->data[s->top]=e;
return 1;
}
}//入栈操作 
 
int Pop(SqStack *s,int e){
if(s->top==-1) return 0;//栈空条件 
else{
e=s->data[s->top];
s->top--;//栈顶下移 
printf("%d已出栈\n",e);
return 1;
}
}//出栈操作

typedef struct{
int front;//队头元素 
int rear;//队尾元素 
int data[MAXSIZE];
}SqQueue ,*linkQueue;//队列初始定义 

linkQueue InitQueue(SqQueue *q){
q=(linkQueue)malloc(sizeof(SqQueue));
q->front=-1;
q->rear=-1;
return q;
} //队列初始化 

int enQueue(SqQueue *q,int e){
if(q->rear==MAXSIZE-1) return 0;//队满条件 
else{
q->rear++;//队尾上移 
q->data[q->rear]=e;
return 1;
}
}//入队操作

int deQueue(SqQueue *q,int e){
if(q->front==q->rear) return 0;//队空条件 
else{
q->front++;//队头上移 
e=q->data[q->front];
printf("%d已出队\n",e);
return 1;
}
} //出队操作

实验内容（附上源代码及必要注释）
数据结构：图G{edges[MAXV][MAXV],n,e,vexs[MAXV]}//邻接矩阵，顶点数，边数
图A{AdjList[MAXV]，n，e}//邻接表，顶点数，边数
AdjList{vexs，first}//存放顶点信息和头结点，不同的顶点信息对应不同的头结点，头结点信息就是该顶点的邻接点信息
ANode{adjvex，ANode* next} //邻接表子表，是一条单链表，以first为头结点，内部存放adjvex为顶点信息
源代码
#include <stdio.h>
#include <stdlib.h>

#define MAXV 100

typedef struct {
    int edges[MAXV][MAXV]; // 邻接矩阵
    int n, e; // 顶点数，边数
    int vexs[MAXV]; // 顶点信息存放
} MatGraph; // 图的建立，用来建立邻接矩阵

typedef struct ANode {
    int adjvex; // 边结点信息
    struct ANode* next;
} ANode; // 邻接表子表的建立

typedef struct VNode {
    int vexs; // 顶点信息存放
    ANode* first; // 第一条边
} VNode; // 邻接表整表

typedef struct {
    VNode* AdjList[MAXV]; // 邻接表
    int n, e; // 顶点数和边的数量
} AdjGraph; // 图的建立，用来建立邻接表

int LocateMat(MatGraph G, int v) {
    for (int i = 0; i < G.n; i++) {
        if (G.vexs[i] == v) return i;
    }
    return -1;
}

void CreateGraph(MatGraph *G) {
    printf("输入图的顶点个数和边的数量\n");
    scanf("%d %d", &G->n, &G->e); // 输入图的顶点个数，边的个数
    for (int i = 0; i < G->n; i++) {
        printf("输入顶点信息\n");
        scanf("%d", &G->vexs[i]);
    }
    for (int i = 0; i < G->n; i++) {
        for (int j = 0; j < G->n; j++) {
            G->edges[i][j] = 0;
        }
    } // 初始化邻接矩阵

    for (int k = 0; k < G->e; k++) {
        int n1, n2;
        printf("输入顶点关系\n");
        scanf("%d %d", &n1, &n2); // 输入两个顶点，如果输入，就在邻接矩阵相应位置填入1
        int v1 = LocateMat(*G, n1);
        int v2 = LocateMat(*G, n2);
        if (v1 != -1 && v2 != -1) {
            G->edges[v1][v2] = 1;
            G->edges[v2][v1] = G->edges[v1][v2]; // 填入对称位置
        }
    }
} // 邻接矩阵创建完毕

void CreateAdjList(AdjGraph *A, MatGraph G) {
    ANode* p; // 结点声明
    *A = (AdjGraph*)malloc(sizeof(AdjGraph));

    (*A)->n = G.n;
    (*A)->e = G.e;
    for (int i = 0; i < (*A)->n; i++) {
        (*A)->AdjList[i] = (VNode*)malloc(sizeof(VNode));
        (*A)->AdjList[i]->vexs = G.vexs[i];
        (*A)->AdjList[i]->first = NULL;
    }

    for (int i = 0; i < (*A)->n; i++) { // 邻接表填入
        for (int j = 0; j < (*A)->n; j++) { // 邻接表子表填入
            if (G.edges[i][j] != 0) {
                p = (ANode*)malloc(sizeof(ANode));
                p->adjvex = G.vexs[j];
                p->next = (*A)->AdjList[i]->first;
                (*A)->AdjList[i]->first = p;
            }
        }
    } // 按照邻接矩阵自动填入邻接表
}

int visited[MAXV] = {0}; // 被访问标记

void DFS(MatGraph G, int v) {
    printf("%d\t", v); // 头输出
    int vi = LocateMat(G, v);
    visited[vi] = 1; // 已被访问过
    for (int i = 0; i < G.n; i++) {
        if (G.edges[vi][i] == 1 && visited[i] == 0) DFS(G, G.vexs[i]); // 递归输出
    }
} // 深度遍历

int LocateAdj(AdjGraph *A, int v) {
    for (int i = 0; i < A->n; i++) {
        if (A->AdjList[i]->vexs == v) return i;
    }
    return -1;
}

void BFS(AdjGraph *A, int v) {
    printf("%d\t", v);
    ANode *p;
    int vi = LocateAdj(A, v);
    for (int i = 0; i < A->n; i++) {
        visited[i]=0;
    }
    visited[vi] = 1;
    int QU[MAXV], front = 0, rear = 0; // 定义一个队列，用来存放输出的顶点
    
    rear = (rear + 1) % MAXV;
    QU[rear] = vi;

    while (front != rear) { // 队空结束输出
        front = (front + 1) % MAXV; // 输出头结点
        vi =QU[front];
        p = A->AdjList[vi]->first; // 定位一下头结点的邻接表
        while (p != NULL) {
            int wi = LocateAdj(A, p->adjvex);
            if (wi != -1 && visited[wi] == 0) {
                printf("%d\t", p->adjvex);
                visited[wi] = 1;
                rear = (rear + 1) % MAXV;
                QU[rear] = wi;
            }
            p = p->next;
        }
    }
} // 广度遍历

int main() {
    MatGraph G;
    AdjGraph* A;
    CreateGraph(&G);
    
    CreateAdjList(&A, G);
    DFS(G, 1);
    printf("\n");
    BFS(A, 1);
    return 0;
}

#define MAXSIZE 100
int r[MAXSIZE]={0};//创建有序表 

typedef struct BSTNode{
int key;//关键词 
struct BSTNode *lchild,*rchild;
}BSTNode; 

BSTNode* InsertBSTNode(BSTNode* bt,int k){
if(bt==NULL){
bt=(BSTNode*)malloc(sizeof(BSTNode));
bt->key=k;
bt->lchild=bt->rchild=NULL;
}//如果结点是空的，直接创建一个新节点 
else if(k<bt->key) bt->lchild=InsertBSTNode(bt->lchild,k); //小于键值，往左构建 
else bt->rchild=InsertBSTNode(bt->rchild,k); //大于键值，往右构建 
return bt; //返回结点 
}//创建结点函数 

BSTNode* CreateBSTtree(int n){

int key;//关键词 

for(int i=0;i<n;i++){
printf("输入键值\n");
scanf("%d",&key);//输入关键词 
bt=InsertBSTNode(bt,key);//插入并创建结点 
}
return bt;//返回根结点 
}//创建排序树

int SearchBSTtree(BSTNode* bt,int k){
while(bt!=NULL){
if(k==bt->key) return 1;
else if(k<bt->key) bt=bt->lchild;
else bt=bt->rchild;
}
return 0;
}//在二叉排序树中查找相关键值

int FIND(int n,int k){

}else if(k<r[mid]) high=mid-1;
else low=mid+1;
}
if(find) return i+1;

} //有序表的二分查找

#include <bits/stdc++.h>

#define MAXSIZE 100
typedef struct{
char data[MAXSIZE];
int length;
}SqString;

void InitSqString(SqString *s){
char x;
while(s->length<=MAXSIZE){
scanf("%c",&x);
if(x!='\n'){
s->data[s->length]=x;
s->length++;
continue;
}
else break;//输入直到换行符，否则一直循环 
}
}

int BF(SqString s,SqString t){
int lens=s.length;
int lent=t.length;
int i=0;
int j=0;
for (; i < lens; i++) {
    for (j=0; j < lent; j++) {
            if (s.data[i + j] != t.data[j]) break;//如果有一位不符合，跳出循环 
        }
        if(j==lent) return 1;//完全符合，则j会多进一位，与lent相等 
}
return -1;
}//BF算法

int* Getnext(SqString t) {
static int next[MAXSIZE]={0};//初始化 
int lent=t.length;
int k=0;
for(int i=0;i<=lent;i++){
if(i==0) next[i]=-1;
else{
if(i==1) next[i]=0;
else{
k=next[i-1];//获得前一项next值
while(k>=0){
if(t.data[k]==t.data[i-1]) {next[i]=k+1;break;}//如果对得上 next+1
else k--;//对不上，就看看k值-1对不对得上
}
}
}
k=0;//循环结束，清空k的值 
}
return next; //输出next整个数组 
}

int KMP(SqString s,SqString t){
int *next;
next=Getnext(t);
int lens=s.length;
int lent=t.length;
int i=0;
int j=0;

while(i<lens){
if(s.data[i]==t.data[j]){
i++;
j++;
if(j==lent) return 1;//如果j值和t长度相同说明t遍历完了，直接结束
}else{
if(next[j]==-1) i++;
else j=next[j];//没对上就使用kmp方法
}
}
return -1;
}
