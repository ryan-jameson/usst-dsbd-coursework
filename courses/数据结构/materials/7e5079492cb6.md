6.写出创建二叉排序树的算法
#include <bits/stdc++.h>
typedef struct BSTNode{
int key;//关键词 
struct BSTNode *lchild,*rchild;
}BSTNode; 

BSTNode* InsertBSTNode(BSTNode* bt,int k){
if(bt==NULL){
bt=(BSTNode*)malloc(sizeof(BSTNode));
bt->key=k;
bt->lchild=bt->rchild=NULL;
}//如果结点是空的，直接创建一个新节点 
else if(k<bt->key) bt->lchild=InsertBSTNode(bt->lchild,k); //小于键值，往左构建 
else bt->rchild=InsertBSTNode(bt->rchild,k); //大于键值，往右构建 
return bt; //返回结点 
}//创建结点函数 

BSTNode* CreateBSTtree(){
BSTNode* bt=NULL;//起始空节点 
int n,key;//关键词数量 和关键词 
printf("输入键值数量\n");
scanf("%d",&n);

for(int i=0;i<n;i++){
printf("输入键值\n");
scanf("%d",&key);//输入关键词 
bt=InsertBSTNode(bt,key);//插入并创建结点 
}
return bt;//返回根结点 
}//创建排序树

void PrintBSTtree(BSTNode* bt){
if (bt == NULL)
return;
    //中序遍历左子树
PrintBSTtree(bt->lchild);
    //访问根结点
printf("%d\t", bt->key);
    //中序遍历右子树
PrintBSTtree(bt->rchild);
} //中序遍历输出排序树，如果是递增序列代表构建成功

main(){
BSTNode* bt;
bt=CreateBSTtree();
PrintBSTtree(bt); 
}
