void SelectSort(Rectype SqList[],int n) {
Rectype tmp;//存放被排序结点的临时结点
for(int i=0;i<n-1;i++){
int t=i;//最小键值标记，如果无序区没有比该键值还小的，就保留这个标记 
for(int j=i+1;j<n;j++){
if(SqList[j].key<SqList[t].key) t=j;
}//找无序区最小
if(t!=i){
tmp=SqList[i];
SqList[i]=SqList[t];
SqList[t]=tmp;
}
printf("第%d趟：",i+1);
for(int k=0;k<n;k++){
printf("%d\t",SqList[k].key);
printf("%c\t",SqList[k].info);
} //输出排序结果 
printf("\n");
} 
}//简单选择排序 
单链表存储
void SelectSort(RecNode* L,int n) {
RecNode* p = L->next; // 第一个数据节点
    RecNode* minNode;

    for (int i = 0; i < n - 1; i++) {
        minNode = p; // 假设当前节点是最小值
        RecNode* pre=p->next;// 移动到下一个节点

        while (pre != NULL) {
            if (pre->data < minNode->data) {
                minNode = pre; // 更新最小值节点
            }
            pre = pre->next;
        }

        if (minNode != p) {
            // 交换p和minNode的值
            int temp = minNode->data;
            minNode->data = p->data;
            p->data = temp;
        }
        p = p->next;//p前面的结点都是已经排序好的 
    }
}//简单选择排序

void InsertSort(Rectype SqList[],int n){
Rectype tmp;//存放被排序结点的临时结点 
for(int i=1;i<n;i++){
if(SqList[i].key<SqList[i-1].key){
tmp=SqList[i];
int j=i-1; //j代表着该结点的前一位 
do{//寻找插入位置 
SqList[j+1]=SqList[j]; //将前一位比较大的后移 
j--;
}while(j>=0 && tmp.key<SqList[j].key); //直到j不能再大或者临时结点的键值比前一位的刚好大 
SqList[j+1]=tmp;//将临时结点存放
}
printf("第%d趟：",i);
for(int k=0;k<n;k++){
printf("%d\t",SqList[k].key);
printf("%c\t",SqList[k].info);
} //输出排序结果 
printf("\n");
} 
} //直接插入排序 
单链表存储
void InsertSort(RecNode *L,int n){

RecNode* p = L->next; // 第一个结点 
    RecNode* tmp;
    
for(int i=0;i<n-1;i++) {
RecNode* pre = L; // 头结点
        tmp = p->next; // 被排序结点 
        if (p->data >tmp->data) { // p的数据大于前一个节点的数据
            // 从头结点开始移动pre
            while (pre->next->data< p->data) {
                pre = pre->next;
            }
            // 将p插入到pre之后
            p->next = pre->next;
            pre->next = p;
        }
    }
    
}//直接插入排序
