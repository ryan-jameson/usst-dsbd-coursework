#include <bits/stdc++.h>
#define MAXSIZE 100

typedef struct {
	int key;
	char info;
}Rectype;
Rectype SqList[MAXSIZE],R[MAXSIZE];

void CreateRec(Rectype SqList[],int n){
	for(int i=0;i<n;i++){
		printf("输入表键值\n");
		scanf("%d",&SqList[i].key);//输入表键值 
		printf("输入表元素\n");
		scanf("%s",&SqList[i].info);//输入表元素 
	} 
	for(int i=0;i<n;i++){
		printf("%d\t",SqList[i].key);
		R[i].key=SqList[i].key;
		printf("%c\t",SqList[i].info);
		R[i].info=SqList[i].info;
	} //输出整个原始表查看是否输入完成 
	printf("\n");
}

void InsertSort(Rectype SqList[],int n){
	Rectype tmp;//存放被排序结点的临时结点 
	for(int i=1;i<n;i++){
		if(SqList[i].key<SqList[i-1].key){
			tmp=SqList[i];
			int j=i-1; //j代表着该结点的前一位 
			do{//寻找插入位置 
				SqList[j+1]=SqList[j]; //将前一位比较大的后移 
				j--;
			}while(j>=0 && tmp.key<SqList[j].key); //直到j不能再大或者临时结点的键值比前一位的刚好大 
			SqList[j+1]=tmp;//将临时结点存放
		}
		printf("第%d趟：",i);
		for(int k=0;k<n;k++){
			printf("%d\t",SqList[k].key);
			printf("%c\t",SqList[k].info);
		} //输出排序结果 
		printf("\n");
	} 
} //直接插入排序 

void SelectSort(Rectype SqList[],int n) {
	Rectype tmp;//存放被排序结点的临时结点
	for(int i=0;i<n-1;i++){
		int t=i;//最小键值标记，如果无序区没有比该键值还小的，就保留这个标记 
		for(int j=i+1;j<n;j++){
			if(SqList[j].key<SqList[t].key) t=j;
		}//找无序区最小
		if(t!=i){
			tmp=SqList[i];
			SqList[i]=SqList[t];
			SqList[t]=tmp;
		}
		printf("第%d趟：",i+1);
		for(int k=0;k<n;k++){
			printf("%d\t",SqList[k].key);
			printf("%c\t",SqList[k].info);
		} //输出排序结果 
		printf("\n");
	} 
}//简单选择排序 

main(){
	int n;//表长
	printf("输入表长\n");
	scanf("%d",&n);//输入表长
	CreateRec(SqList,n);
	printf("直接插入排序\n");
	InsertSort(R,n); 
	printf("简单选择排序\n");
	SelectSort(SqList,n);
}
